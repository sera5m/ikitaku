cmake_minimum_required(VERSION 3.16)
project(ikitaku LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(IKITAKU_STANDALONE "Trace host only" ON)
set(VULCAN_LANG_DIR "" CACHE PATH "sera5m/vulcan-lang")
add_library(ikitaku_engine engine/ikitaku.cpp)
target_include_directories(ikitaku_engine PUBLIC engine)
if(IKITAKU_STANDALONE)
  target_compile_definitions(ikitaku_engine PUBLIC IKITAKU_STANDALONE)
endif()
add_executable(iki_trace host/iki_trace.cpp)
target_link_libraries(iki_trace PRIVATE ikitaku_engine)
